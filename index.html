const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTRqMqbCRZdfrIDgdkarWSH7iy3qhwhSCsfjAOcKVYO60m3smkMLMIT8hQWBdmrboORQJGsC3FwmVRq/pub?gid=389737636&single=true&output=csv"; // ⚡ 1. This executes INSTANTLY (under 200ms) because it just reads the ready-made cache function doGet() { const cache = CacheService.getScriptCache(); let cachedData = cache.get("roblox_processed_data"); // Fallback if cache expires or is empty if (!cachedData) { cachedData = JSON.stringify(getGamesData()); cache.put("roblox_processed_data", cachedData, 21600); // Save for 6 hours } return ContentService .createTextOutput(cachedData) .setMimeType(ContentService.MimeType.JSON); } // ⚡ 2. This will run automatically in the background via a time-driven trigger function cacheGamesData() { try { const freshData = JSON.stringify(getGamesData()); const cache = CacheService.getScriptCache(); cache.put("roblox_processed_data", freshData, 21600); Logger.log("Cache successfully updated!"); } catch(e) { Logger.log("Error updating cache: " + e.toString()); } } // This handles the heavy lifting away from the user's loading screen function getGamesData() { const res = UrlFetchApp.fetch(CSV_URL); const text = res.getContentText(); const rows = text .trim() .split("\n") .map(r => r.split(",").map(x => x.replace(/"/g, "").trim())) .slice(1); const games = {}; const uniqueIds = new Set(); for (const r of rows) { const id = r[0]; const mins = Number(r[2]) || 0; if (!id) continue; uniqueIds.add(id); let date = new Date(); if (r[3]) { const parts = r[3].split(" "); if (parts.length >= 2) { const datePart = parts[0].split("/"); const timePart = parts[1].split(":"); const month = parseInt(datePart[0]) - 1; const day = parseInt(datePart[1]); const year = parseInt(datePart[2]); const hour = parseInt(timePart[0] || 0); const minute = parseInt(timePart[1] || 0); const second = parseInt(timePart[2] || 0); date = new Date(year, month, day, hour, minute, second); } } const dateKey = `${date.getFullYear()}-${date.getMonth()}-${date.getDate()}`; const gameKey = `${id}_${dateKey}`; if (!games[gameKey]) { games[gameKey] = { id, minutes: 0, date }; } games[gameKey].minutes += mins; } const idArray = Array.from(uniqueIds); const apiMetaData = {}; if (idArray.length > 0) { try { const idString = idArray.slice(0, 100).join(","); const gameRes = UrlFetchApp.fetch(`https://games.roblox.com/v1/games?universeIds=${idString}`); const gameJson = JSON.parse(gameRes.getContentText()); const thumbRes = UrlFetchApp.fetch(`https://thumbnails.roblox.com/v1/games/icons?universeIds=${idString}&size=512x512&format=Png&isCircular=false`); const thumbJson = JSON.parse(thumbRes.getContentText()); if (gameJson.data) { gameJson.data.forEach(g => { apiMetaData[g.id] = { name: g.name, image: "" }; }); } if (thumbJson.data) { thumbJson.data.forEach(t => { if (apiMetaData[t.targetId]) apiMetaData[t.targetId].image = t.imageUrl || ""; }); } } catch (e) { Logger.log("Roblox API Error: " + e.toString()); } } const sorted = Object.entries(games).sort((a, b) => b[1].minutes - a[1].minutes); const output = []; for (const [key, g] of sorted) { const meta = apiMetaData[g.id] || { name: "Unknown Game", image: "" }; output.push({ id: g.id, name: meta.name, minutes: g.minutes, date: g.date.toISOString(), image: meta.image }); } return output; }
